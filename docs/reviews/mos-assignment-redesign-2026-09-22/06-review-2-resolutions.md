-- Atomic conditional enqueue. No 40001; conflicts are WS409 elsewhere, this one never conflicts (it de-dupes).
CREATE OR REPLACE FUNCTION public.mos_meta_ad_enqueue(
  p_event_id uuid, p_content_id uuid, p_content_title text,
  p_ad_row_id uuid, p_execution_id uuid, p_ad_set_id uuid,
  p_platform_adset_id text, p_ad_set_name text,
  p_approved_by_auth_uid uuid, p_approved_by_user_id uuid, p_phase text
) RETURNS jsonb LANGUAGE plpgsql SECURITY DEFINER AS $$
DECLARE v_job uuid := public.mos_meta_job_id(p_event_id);
        v_row uuid := COALESCE(p_ad_row_id, public.mos_meta_ad_row_id(p_event_id));
        v_auto jsonb := jsonb_build_object('state','queued','phase',p_phase,'job_id',v_job::text,
                          'queued_at', now(), 'approved_by_user_id', p_approved_by_user_id, 'error', null);
        v_inserted boolean := false;
        v_state text;
BEGIN
  -- (1) ensure the ad row exists (deterministic id when no placeholder)
  IF p_ad_row_id IS NULL THEN
    INSERT INTO public.mos_execution_ads (id, execution_id, ad_set_id, content_id, label, status, creative)
    VALUES (v_row, p_execution_id, p_ad_set_id, p_content_id, p_content_title, 'waiting',
            jsonb_build_object('auto_ad', v_auto))
    ON CONFLICT (id) DO NOTHING;
  END IF;

  -- (2) insert the job idempotently; v_inserted = did THIS call create it?
  INSERT INTO public.generation_jobs (id, record_id, message_id, generation_id, user_id, kind, status, prompt, params)
  VALUES (v_job, p_content_id, v_row::text, NULL, p_approved_by_auth_uid, 'meta-ad', 'queued', NULL,
          jsonb_build_object('content_id',p_content_id,'ad_row_id',v_row,'execution_id',p_execution_id,
            'ad_set_id',p_ad_set_id,'platform_adset_id',p_platform_adset_id,'ad_set_name',p_ad_set_name,
            'approved_by_user_id',p_approved_by_user_id,'phase',p_phase,'event_id',p_event_id))
  ON CONFLICT (id) DO NOTHING;
  GET DIAGNOSTICS v_inserted = ROW_COUNT;  -- 1 = brand new, 0 = already existed

  -- (3) reset the ad row ONLY when the job is genuinely new; else leave worker progress intact
  IF v_inserted THEN
    UPDATE public.mos_execution_ads
       SET content_id = p_content_id, ad_set_id = p_ad_set_id, status = 'waiting',
           creative = jsonb_set(COALESCE(creative,'{}'::jsonb), '{auto_ad}', v_auto), updated_at = now()
     WHERE id = v_row;
  END IF;

  SELECT creative->'auto_ad'->>'state' INTO v_state FROM public.mos_execution_ads WHERE id = v_row;
  RETURN jsonb_build_object('enqueued', v_inserted, 'job_id', v_job, 'ad_row_id', v_row, 'ad_state', v_state);
END $$;

PLAIN: The reviewer is right. Today, when a manager approves an ad, the code does two separate things: first it flips the ad's little status card back to 'queued/waiting', then it drops a job on the worker's queue. If something goes wrong and we retry that approval later (a recovery pass, or the manager clicking again after a lost response), the retry flips the status card back to 'queued' AGAIN — even if the worker has already started building the ad or already created it on Meta — and then finds the job already exists and adds nothing. Result: the ad gets 'reset' to a waiting state that nobody is working on, and it just sits there stuck. Giving the job a fixed, predictable id (so we never create two jobs) is a good and necessary fix, but on its own it does NOT stop that status-card reset from wrecking a job the worker already owns.\n\nThe fix is to make the whole thing one indivisible step: a single database function that (a) creates the job only if it does not already exist, and (b) resets the status card ONLY when it actually created the job for the first time. On every retry after that, it looks, sees the job is already there, and leaves everything alone. So you can safely retry it as many times as you want.\n\nThe second problem: after the approval, the code also sends notifications and kicks off the ad. But the notification helper quietly hides its own errors, and the database calls report failure by returning an error object rather than crashing — so the code can 'think' a step succeeded when it actually failed, and mark it 'done'. The fix is a rule: mark a step 'done' only after we have actually confirmed it worked, not just because we called it. A step that failed temporarily stays 'pending' so a background retry (every 1-2 minutes on busy launch days) picks it up; a step that can never work is marked 'failed' and shown to a human — it is never silently swallowed. This matches the house rules: never lie about success, never hide an error.

CREATE OR REPLACE FUNCTION public.mos_capacity_check(p_reservations jsonb)
RETURNS jsonb LANGUAGE sql STABLE SECURITY DEFINER SET search_path TO 'public' AS $$
  WITH prop AS (
    SELECT (x->>'assignee_user_id')::uuid uid,
           COALESCE(x->>'bucket','post') bucket,
           (x->>'planned_start')::date ps,
           COALESCE((x->>'weight')::numeric,1) total,
           COALESCE(NULLIF(x->>'spread',''),'per_day') spread
      FROM jsonb_array_elements(COALESCE(p_reservations,'[]'::jsonb)) x
     WHERE NULLIF(x->>'assignee_user_id','') IS NOT NULL
  ), spread AS (
    SELECT prop.uid, prop.bucket, s.day, s.weight w
      FROM prop CROSS JOIN LATERAL
           public.mos_spread_effort_mode(GREATEST(prop.ps, public.mos_perf_today()),
                                         prop.total, prop.spread='same_day') s
  ), agg AS (SELECT uid,bucket,day,sum(w) proposed FROM spread GROUP BY 1,2,3)
  SELECT COALESCE(jsonb_agg(jsonb_build_object(
           'user_id',a.uid,'day',a.day,'bucket',a.bucket,
           'existing',x.existing,'proposed',a.proposed,
           'capacity',public.mos_user_daily_slots(a.uid,a.bucket),
           'on_leave',public.mos_on_leave(a.uid,a.day))),'[]'::jsonb)
    FROM agg a CROSS JOIN LATERAL
         (SELECT COALESCE(sum(l.weight),0) existing FROM public.mos_work_ledger_v l
           WHERE l.user_id=a.uid AND l.day=a.day AND l.bucket=a.bucket) x
   WHERE public.mos_on_leave(a.uid,a.day)
      OR (x.existing + a.proposed) > public.mos_user_daily_slots(a.uid,a.bucket);
$$;

CREATE OR REPLACE FUNCTION public.mos_retire_superseded(
  p_campaign_ids uuid[], p_keep_plan_ids uuid[], p_by_plan uuid)
RETURNS void LANGUAGE plpgsql SECURITY DEFINER SET search_path TO 'public' AS $$
DECLARE v_repl uuid[];
BEGIN
  SELECT array_agg(id) INTO v_repl FROM public.mos_campaign_plans
   WHERE campaign_id = ANY(p_campaign_ids)
     AND status IN ('proposed','approved') AND NOT (id = ANY(p_keep_plan_ids));
  IF v_repl IS NULL THEN RETURN; END IF;
  -- 1a unconsumed reservations
  UPDATE public.mos_task_reservations
     SET status='superseded', superseded_by_plan_id=p_by_plan, superseded_at=now(), updated_at=now()
   WHERE plan_id = ANY(v_repl) AND status IN ('reserved','stale') AND consumed_task_id IS NULL;
  -- 1b withdraw offered / bound-unassigned content tasks (NOT carried-forward row tasks)
  UPDATE public.workflow_role_tasks t
     SET status='skipped', closed_at=now(),
         note=COALESCE(t.note,'')||' plan_superseded:'||p_by_plan
   WHERE t.status='open' AND t.assignee_user_id IS NULL AND t.subject_table='mos_content'
     AND t.subject_id IN (SELECT content_id FROM public.mos_content_plan WHERE plan_id = ANY(v_repl));
  UPDATE public.mos_task_reservations r
     SET status='superseded', superseded_by_plan_id=p_by_plan, superseded_at=now(), updated_at=now()
   WHERE r.plan_id = ANY(v_repl) AND r.status='bound' AND r.consumed_task_id IN
         (SELECT id FROM public.workflow_role_tasks WHERE status='skipped');
  -- 1c consumed-behind-assigned: untouched.  1d flip plan status
  UPDATE public.mos_campaign_plans
     SET status='superseded', superseded_by=p_by_plan, updated_at=now() WHERE id = ANY(v_repl);
END $$;

-- COMMIT_MONTH: phased. lock/hash → RETIRE(all) → ONE guard(union) → insert(batch mode)
CREATE OR REPLACE FUNCTION public.mos_campaign_plan_commit_month(
  p_plans jsonb, p_expected_hash text, p_actor uuid)
RETURNS jsonb LANGUAGE plpgsql SECURITY DEFINER SET search_path TO 'public' AS $$

BEGIN
  IF auth.uid() IS NOT NULL AND NOT public.wassell_mos_can('approve_plan') THEN
    RAISE EXCEPTION 'MOS:NOT_ALLOWED' USING ERRCODE='insufficient_privilege'; END IF;
  PERFORM public.mos_ledger_lock();                                    -- PHASE 0

    SELECT COALESCE(NULLIF(e->'materialise'->>'campaign_id','')::uuid,

    IF (SELECT status FROM public.mos_campaign_plans WHERE id=v_pid)='proposed' THEN
      IF v_camp IS NULL THEN
        RAISE EXCEPTION 'MOS:PLAN_CAMPAIGN_UNRESOLVED %',v_pid USING ERRCODE='invalid_parameter_value'; END IF;
      v_camps := v_camps || v_camp;
      v_union := v_union || COALESCE(e->'reservations','[]'::jsonb);

  v_cells := public.mos_capacity_check(v_union);                       -- PHASE 2 (single guard)
  IF jsonb_array_length(v_cells) > 0 THEN
    RAISE EXCEPTION 'capacity_conflict' USING ERRCODE='WS409', DETAIL=v_cells::text; END IF;
  FOR e IN SELECT * FROM jsonb_array_elements(COALESCE(p_plans,'[]'::jsonb)) LOOP  -- PHASE 3

               COALESCE(e->'materialise','{}'::jsonb), p_actor,

    v_n := v_n + 1;
  END LOOP;
  RETURN jsonb_build_object('ok',true,'plans',v_n,'snapshot_hash',v_hash,'committed',v_out);
END $$;

PLAIN: The plan checks whether a new schedule would overload anyone BEFORE it removes the old schedule it is replacing. So when you swap seven pieces of work for seven equivalent ones, the system briefly sees fourteen and refuses the swap — a cap of seven looks blown even though nothing really changed. The reviewer is right; I confirmed it in the code and against the live database.

The fix is to do the steps in one strict order inside a single all-or-nothing operation: (1) lock the books; (2) first retire every old booking that is being replaced — but leave alone any work someone has already started; (3) only then count the total new load against each person's daily limit, once; (4) then write the new bookings. Because retirement happens before counting, the old sevens are gone when the new sevens are measured, so 7-for-7 fits. If any step fails, everything rewinds — the retired bookings come back untouched.

Two safety points the operator cares about: work that is already assigned and in progress is never quietly retired or double-booked over — if a replacement would land on top of it, the system stops and says so, so nobody's in-flight task gets clobbered. CREATE OR REPLACE FUNCTION public.mos_publication_risk(
  p_subject_table text, p_subject_id uuid)
RETURNS TABLE(projected_ready_at timestamptz, required_ready_at timestamptz,
              publish_at timestamptz, risk text, risk_step text, risk_reason text)
LANGUAGE plpgsql STABLE SECURITY DEFINER SET search_path TO 'public' AS $$
DECLARE
  v_now      timestamptz := clock_timestamp();
  v_open     public.workflow_role_tasks%ROWTYPE;
  v_steps    jsonb; v_pos int; v_cursor timestamptz;
  v_is_paid  boolean; v_lead_h numeric; v_A timestamptz;
  v_ad       record; v_pub record; v_daystart timestamptz;
  r record; v_started boolean := false;
BEGIN
  v_is_paid := p_subject_table='mos_content'
    AND EXISTS (SELECT 1 FROM mos_content c WHERE c.id=p_subject_id AND c.purpose='paid');

  -- required_ready_at / publish_at instants (unchanged from v3 §1c).
  IF v_is_paid THEN
    SELECT (rc.refresh_on::timestamp AT TIME ZONE 'Asia/Riyadh') INTO v_A
      FROM mos_creative_slots s JOIN mos_refresh_cycles rc ON rc.id=s.cycle_id
     WHERE s.content_id=p_subject_id ORDER BY rc.refresh_on LIMIT 1;
    v_lead_h := public.mos_setting_num('planning.ad_build_lead_hours',2);
    required_ready_at := v_A - make_interval(hours=>v_lead_h::int);
    publish_at        := v_A + interval '1 day';         -- end of the activation day
  ELSE
    SELECT min(p.planned_at) INTO publish_at FROM mos_publications p JOIN mos_content c ON c.id=p.content_id
     WHERE (CASE WHEN p_subject_table='mos_content_rows' THEN c.row_id=p_subject_id ELSE c.id=p_subject_id END);
    v_lead_h := public.mos_setting_num('planning.release_lead_hours',2);
    required_ready_at := publish_at - make_interval(hours=>v_lead_h::int);
  END IF;

  -- (a) CURRENT STEP = the single OPEN task (one-open index). No done-scan.
  SELECT * INTO v_open FROM public.workflow_role_tasks
   WHERE subject_table=p_subject_table AND subject_id=p_subject_id AND status='open'
   LIMIT 1;

  IF FOUND THEN
    IF v_open.blocked IS TRUE THEN
      RETURN QUERY SELECT NULL::timestamptz, required_ready_at, publish_at,
        'blocked', v_open.step_key, COALESCE(v_open.blocked_reason,'blocked'); RETURN;
    END IF;
    SELECT COALESCE(r.workflow_version_id, c.workflow_version_id)
      INTO v_open.workflow_version_id            -- reuse a scratch var conceptually
      FROM (SELECT 1) x
      LEFT JOIN mos_content_rows r ON p_subject_table='mos_content_rows' AND r.id=p_subject_id
      LEFT JOIN mos_content c      ON p_subject_table='mos_content'      AND c.id=p_subject_id;
    SELECT (v.definition->'metadata'->'steps') INTO v_steps
      FROM workflow_versions v WHERE v.id=v_open.workflow_version_id;
    SELECT ord-1 INTO v_pos FROM jsonb_array_elements(v_steps) WITH ORDINALITY e(el,ord)
     WHERE el->>'key'=v_open.step_key LIMIT 1;

    -- current step c
    v_daystart := (SELECT (scheduled_start::timestamp AT TIME ZONE 'Asia/Riyadh')
                     FROM mos_task_reservations WHERE id=v_open.reservation_id);
    v_cursor := GREATEST(v_now, public.mos_work_due_at(
                  COALESCE(v_open.assigned_at, GREATEST(v_now, COALESCE(v_daystart,v_now))),
                  public.mos_step_target_hours(v_open.step_key)));
    risk_step := v_open.step_key;

    -- remaining steps: pinned positions > v_pos, THIS round, done rows ignored.
    FOR r IN SELECT el->>'key' step_key, ord-1 pos
               FROM jsonb_array_elements(v_steps) WITH ORDINALITY e(el,ord)
              WHERE ord-1 > v_pos ORDER BY ord
    LOOP
      v_daystart := (SELECT (res.scheduled_start::timestamp AT TIME ZONE 'Asia/Riyadh')
                       FROM mos_task_reservations res
                      WHERE res.content_id=p_subject_id AND res.step_key=r.step_key
                      ORDER BY res.scheduled_start LIMIT 1);
      v_cursor := public.mos_work_due_at(GREATEST(v_cursor, COALESCE(v_daystart,v_cursor)),
                                         public.mos_step_target_hours(r.step_key));
    END LOOP;

    projected_ready_at := v_cursor;
    risk := CASE WHEN projected_ready_at<=required_ready_at THEN 'on_track'
                 WHEN publish_at IS NULL OR projected_ready_at<=publish_at THEN 'at_risk'
                 ELSE 'late' END;
    RETURN NEXT; RETURN;
  END IF;

  -- (b) TERMINAL: no open task. Never fall through to on_track.
  IF v_is_paid THEN
    SELECT ea.status, ea.activated_at, ea.platform_ad_id,
           ea.creative->'auto_ad' auto_ad INTO v_ad
      FROM mos_creative_slots s JOIN mos_execution_ads ea ON ea.id=s.ad_row_id
     WHERE s.content_id=p_subject_id ORDER BY s.activate_on LIMIT 1;
    projected_ready_at := NULL; risk_step := 'activation';
    IF v_ad.activated_at IS NOT NULL OR v_ad.auto_ad->>'ad_status'='ACTIVE' THEN
      risk := 'active';                                   -- terminal OK
    ELSIF v_ad.platform_ad_id IS NOT NULL AND v_now < publish_at THEN
      risk := 'on_track';                                 -- built, awaiting tick
    ELSIF v_now < required_ready_at THEN
      risk := 'on_track';
    ELSIF v_now < publish_at THEN
      risk := 'at_risk'; risk_reason := 'ad not built yet';
    ELSE
      risk := 'late'; risk_reason := 'activation day passed, ad not live';
    END IF;
  ELSE
    SELECT bool_and(p.status='published') all_pub, bool_or(p.status IN ('planned','draft')) any_planned,
           bool_or(p.status='scheduled') any_sched, bool_or(p.bundle_error IS NOT NULL) any_err,
           count(*) n INTO v_pub
      FROM mos_publications p JOIN mos_content c ON c.id=p.content_id
     WHERE (CASE WHEN p_subject_table='mos_content_rows' THEN c.row_id=p_subject_id ELSE c.id=p_subject_id END);
    projected_ready_at := NULL; risk_step := 'publication';
    IF v_pub.n=0 THEN risk:='late'; risk_reason:='no publication row';
    ELSIF v_pub.all_pub THEN risk:='published';
    ELSIF v_pub.any_err THEN risk:='late'; risk_reason:='publish failed';
    ELSIF v_pub.any_planned AND publish_at < v_now THEN risk:='late'; risk_reason:='never scheduled';
    ELSIF v_pub.any_sched AND v_now <= publish_at THEN risk:='on_track';
    ELSE risk:='late'; risk_reason:='scheduled but past publish time'; END IF;
  END IF;
  RETURN NEXT;
END $$;
