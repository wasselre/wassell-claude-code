# [commit-guard] Corrected-guard invariant and SQL skeleton

**Invariant** — under mos_ledger_lock(), for every (assignee A, working-day D, bucket B) touched by (ledger existing ∪ this commit's new bookings):  existing(A,D,B) + proposed(A,D,B) ≤ mos_user_daily_slots(A,B), and A not on leave on D.  existing = mos_work_ledger_v (open tasks spread as the ledger does + reserved-unconsumed reservations + stale + manual; consumed reservations already excluded, their task counted — each unit once).  proposed spread by the SAME rule the ledger uses: same-day IFF the reservation is a ROW.  Otherwise raise WS409 'capacity_conflict' with the offending (person, day, bucket, existing, proposed, cap).

**Fix 1 — one-line parity in the proposed CTE (both commit and commit_month):**
```sql
-- was: COALESCE(NULLIF(x->>'spread',''),'per_day') AS spread  ... prop.spread='same_day'
-- now derive same-day structurally, matching mos_work_ledger_v's row_id IS NOT NULL:
(NULLIF(x->>'row_key','') IS NOT NULL) AS same_day
...
CROSS JOIN LATERAL public.mos_spread_effort_mode(
      GREATEST(prop.ps, public.mos_perf_today()), prop.total, prop.same_day) s
```

**Fix 2 — whole-month union guard in commit_month, after the lock+hash gate, BEFORE the per-plan loop:**
```sql
WITH allres AS (
  SELECT x FROM jsonb_array_elements(COALESCE(p_plans,'[]'::jsonb)) pl
  CROSS JOIN LATERAL jsonb_array_elements(COALESCE(pl->'reservations','[]'::jsonb)) x),
prop AS (
  SELECT (x->>'assignee_user_id')::uuid uid, COALESCE(x->>'bucket','post') bucket,
         (x->>'planned_start')::date ps, COALESCE((x->>'weight')::numeric,1) total,
         (NULLIF(x->>'row_key','') IS NOT NULL) same_day
    FROM allres WHERE NULLIF(x->>'assignee_user_id','') IS NOT NULL),
spread AS (SELECT uid,bucket,s.day,s.weight w FROM prop
  CROSS JOIN LATERAL public.mos_spread_effort_mode(GREATEST(ps,public.mos_perf_today()),total,same_day) s),
agg AS (SELECT uid,bucket,day,sum(w) proposed FROM spread GROUP BY 1,2,3)
SELECT COALESCE(jsonb_agg(jsonb_build_object('user_id',a.uid,'day',a.day,'bucket',a.bucket,
         'existing',x.existing,'proposed',a.proposed,'cap',public.mos_user_daily_slots(a.uid,a.bucket))),'[]'::jsonb)
  INTO v_cells FROM agg a
  CROSS JOIN LATERAL (SELECT COALESCE(sum(l.weight),0) existing FROM public.mos_work_ledger_v l
     WHERE l.user_id=a.uid AND l.day=a.day AND l.bucket=a.bucket) x
 WHERE public.mos_on_leave(a.uid,a.day)
    OR (x.existing + a.proposed) > public.mos_user_daily_slots(a.uid,a.bucket);
IF jsonb_array_length(v_cells) > 0 THEN
  RAISE EXCEPTION 'capacity_conflict' USING ERRCODE='WS409', DETAIL=v_cells::text; END IF;
```
Evaluated once, before any insert, existing = pre-confirm ledger (no double-count with the about-to-insert rows); union of all plans' proposed → the (سارة,09-21,post) cell = 0+9 > 7 → WS409, order-independent. NOTE: exclude from `existing` the reserved reservations of plans this commit SUPERSEDES (same campaign_id, status in proposed/approved) so a re-preview→re-confirm does not double-count its own prior reservations.

```sql
-- (1) APPLY ONLY: no holder pick, no capacity WHERE. Refill decides the person.
CREATE FUNCTION mos_task_assign_apply(p_task_id uuid, p_user uuid, p_notify boolean)
RETURNS void LANGUAGE plpgsql SECURITY DEFINER SET search_path=public AS $$
DECLARE v workflow_role_tasks%ROWTYPE; v_allow numeric; v_due timestamptz;
BEGIN
  SELECT * INTO v FROM workflow_role_tasks WHERE id=p_task_id FOR UPDATE;
  IF NOT FOUND OR v.status<>'open' OR v.assigned_at IS NOT NULL THEN RETURN; END IF;
  v_allow := COALESCE((SELECT allowance_hours FROM mos_step_rules WHERE step_key=v.step_key),24);
  v_due  := mos_work_due_at(clock_timestamp(), v_allow);  -- + approved-leave extension (as today)
  UPDATE workflow_role_tasks
     SET assignee_user_id=p_user, assigned_at=clock_timestamp(), due_at=v_due,
         waiting_since=NULL, waiting_reason=NULL
   WHERE id=p_task_id;
  IF p_notify THEN PERFORM mos_task_notify_assigned(p_task_id); END IF;
END $$;

-- (2) CREATE-AND-BIND ONLY: no dispatch. Used by refill AND path_start/advance/revise.
CREATE FUNCTION mos_task_open(p_subject_table text, p_subject_id uuid) RETURNS uuid
LANGUAGE plpgsql SECURITY DEFINER SET search_path=public AS $$
DECLARE v_ver uuid; v_steps jsonb; v_first jsonb; v_new uuid;
BEGIN
  PERFORM mos_ledger_lock();  -- reentrant; already held inside refill
  IF EXISTS (SELECT 1 FROM workflow_role_tasks
              WHERE subject_table=p_subject_table AND subject_id=p_subject_id AND status='open')
     THEN RETURN NULL; END IF;
  -- resolve pinned steps off the subject (identical to path_start) ...
  v_first := v_steps->0;
  INSERT INTO workflow_role_tasks(subject_table,subject_id,workflow_version_id,step_key,role_key,round,due_at)
  VALUES (p_subject_table,p_subject_id,v_ver,v_first->>'key',v_first->>'role_key',1,
          now()+COALESCE((v_first->>'due_days')::int,2)*interval '1 day')
  RETURNING id INTO v_new;
  PERFORM mos_plan_bind_reservation(v_new);  -- consume_reservation MINUS the dispatch call
  RETURN v_new;                              -- NO assignment here
END $$;

-- (3) REFILL: sole decider, re-entrancy-guarded, one pass.
CREATE FUNCTION mos_refill() RETURNS jsonb
LANGUAGE plpgsql SECURITY DEFINER SET search_path=public AS $$
DECLARE r record; v_pick uuid;
BEGIN
  IF current_setting('mos.in_refill', true) = '1' THEN
     PERFORM set_config('mos.refill_again','1',true); RETURN jsonb_build_object('reentrant',true);
  END IF;
  PERFORM set_config('mos.in_refill','1',true);
  PERFORM mos_ledger_lock();
  LOOP
    PERFORM set_config('mos.refill_again','0',true);
    -- a. release strays (as mos_dispatch_sweep's first loop does today)
    -- b. OPEN due subjects + refresh cycles via mos_task_open (NO assign)
    -- c. batch = open+unassigned tasks whose deps are satisfied, ordered
    --    publish_at NULLS LAST, publish_rank, waiting_since; PLUS one future
    --    planned day's WHOLE ready batch (skip a fully capacity-blocked
    --    earliest future day so later executable reservations are not hidden)
    FOR r IN (/* the batch */) LOOP
      SELECT h.uid INTO v_pick FROM mos_role_routine_holders(r.role_key) h
       WHERE /* strict capacity vs a RUNNING per-holder tally counting tasks
                AND consumed reservations exactly once across every occupied day */
       ORDER BY /* least-recently-assigned */ LIMIT 1;
      IF v_pick IS NOT NULL THEN PERFORM mos_task_assign_apply(r.id, v_pick, true); END IF;
    END LOOP;
    EXIT WHEN current_setting('mos.refill_again',true) <> '1';
  END LOOP;
  PERFORM set_config('mos.in_refill','0',true);
  RETURN jsonb_build_object('ok',true);
END $$;
-- path_start/advance/revise: call mos_task_open(...), then request one drain:
--   IF current_setting('mos.in_refill',true)='1' THEN PERFORM set_config('mos.refill_again','1',true);
--   ELSE PERFORM mos_refill(); END IF;   -- never recurse

-- (4) IDEMPOTENT COMPLETION
CREATE TABLE mos_completion_events (
  event_id uuid PRIMARY KEY, task_id uuid NOT NULL,
  subject_table text NOT NULL, subject_id uuid NOT NULL,
  result text NOT NULL, outcome jsonb NOT NULL, created_at timestamptz NOT NULL DEFAULT now());

-- advance gains p_task_id (required) + p_event_id (required):
-- workflow_advance_role_path(p_subject_table,p_subject_id,p_result,p_task_id uuid,p_event_id uuid, ...)
  PERFORM mos_ledger_lock();
  SELECT outcome INTO v_prev FROM mos_completion_events WHERE event_id=p_event_id;
  IF FOUND THEN RETURN v_prev; END IF;                    -- replay
  SELECT * INTO v_task FROM workflow_role_tasks
   WHERE id=p_task_id AND subject_table=p_subject_table AND subject_id=p_subject_id
   FOR UPDATE;                                             -- pin the EXPECTED task
  IF NOT FOUND OR v_task.status<>'open' THEN
     RAISE EXCEPTION 'MOS:TASK_ALREADY_CLOSED' USING ERRCODE='invalid_parameter_value';
  END IF;                                                  -- never touch the successor
  /* ... existing transition, then: open+bind next via mos_task_open + request refill ... */
  INSERT INTO mos_completion_events(event_id,task_id,subject_table,subject_id,result,outcome)
  VALUES (p_event_id,p_task_id,p_subject_table,p_subject_id,p_result,v_return); -- PK backstops dup
  RETURN v_return;
-- content_revise: add p_event_id + take mos_ledger_lock (today it takes only content-row FOR UPDATE).
-- workflow_role_task_transfer: add p_event_id; on replay short-circuit so the note is not appended twice.
```

# [deadline-model] SQL skeleton — mos_plan_chain(subject) : set-returning handoff chain (folds by sort_order; robust to planned_start inversion)

```sql
-- READ-ONLY design skeleton. Not applied.
CREATE OR REPLACE FUNCTION public.mos_plan_chain(
  p_subject_table text, p_subject_id uuid)
RETURNS TABLE(step_key text, sort_order int, booked_day date,
              target_hours numeric, allowance_hours numeric,
              plan_handoff_at timestamptz, plan_due_at timestamptz)
LANGUAGE plpgsql STABLE SECURITY DEFINER SET search_path TO 'public' AS $$
DECLARE
  r          record;
  v_prev_due timestamptz;   -- cand_k = work_due_at(prev_handoff, prev_target)
  v_handoff  timestamptz;
  v_daystart timestamptz;
BEGIN
  FOR r IN
    SELECT res.step_key,
           COALESCE(sr.sort_order, 999) AS so,
           COALESCE(res.planned_start, public.mos_perf_today()) AS bday,
           COALESCE(sr.target_hours, sr.allowance_hours, 4)     AS tgt,
           COALESCE(sr.allowance_hours, 24)                     AS allw
      FROM public.mos_task_reservations res
      JOIN public.mos_step_rules sr ON sr.step_key = res.step_key
     WHERE (CASE WHEN p_subject_table = 'mos_content_rows'
                 THEN res.row_id = p_subject_id
                 ELSE res.content_id = p_subject_id END)
       AND res.status IN ('reserved','stale','consumed')
     ORDER BY so                       -- STEP ORDER, never planned_start
  LOOP
    -- day start of the booked day, snapped forward off a day-off just in case
    v_daystart := (r.bday::timestamp AT TIME ZONE 'Asia/Riyadh');
    IF v_prev_due IS NULL THEN
      v_handoff := v_daystart;                       -- handoff_1 = 00:00 booked
    ELSE
      v_handoff := GREATEST(v_daystart, v_prev_due); -- max(bookedDay, prev+target)
    END IF;
    step_key := r.step_key; sort_order := r.so; booked_day := r.bday;
    target_hours := r.tgt; allowance_hours := r.allw;
    plan_handoff_at := v_handoff;
    plan_due_at     := public.mos_work_due_at(v_handoff, r.allw);
    v_prev_due      := public.mos_work_due_at(v_handoff, r.tgt); -- cand for next
    RETURN NEXT;
  END LOOP;
END $$;
```
Populates mos_task_reservations.planned_handoff_at/planned_due_at at commit and workflow_role_tasks.plan_handoff_at/plan_due_at at consume. Prefer *_corrected over original where present.


# [deadline-model] SQL skeleton — mos_publication_risk(subject) : forecast vs required-ready / publish

```sql
-- READ-ONLY design skeleton. Not applied.
CREATE OR REPLACE FUNCTION public.mos_publication_risk(
  p_subject_table text, p_subject_id uuid)
RETURNS TABLE(projected_ready_at timestamptz, required_ready_at timestamptz,
              publish_at timestamptz, risk text, risk_step text)
LANGUAGE plpgsql STABLE SECURITY DEFINER SET search_path TO 'public' AS $$
DECLARE
  v_anchor  timestamptz;  -- current earliest-open step's ACTUAL handoff, else plan
  v_cursor  timestamptz;
  v_step    text; v_started boolean := false; r record;
BEGIN
  -- publish instant (18:00 etc.) and required-ready = last step's plan_due_at
  SELECT min(planned_at) INTO publish_at FROM public.mos_publications p
    WHERE p.content_id = p_subject_id;              -- rows: resolve via members
  SELECT max(plan_due_at) INTO required_ready_at
    FROM public.mos_plan_chain(p_subject_table, p_subject_id);
  -- fold remaining targets from the current open step, anchored on its actual
  -- handoff (predecessor closed_at) when known, else its plan_handoff_at.
  FOR r IN SELECT * FROM public.mos_plan_chain(p_subject_table, p_subject_id)
           ORDER BY sort_order LOOP
    IF NOT v_started THEN
      -- first not-yet-done step is the anchor
      v_anchor := COALESCE(
        (SELECT t.opened_at FROM public.workflow_role_tasks t
          WHERE t.subject_table=p_subject_table AND t.subject_id=p_subject_id
            AND t.step_key=r.step_key AND t.status='open' LIMIT 1),
        r.plan_handoff_at);
      v_cursor := v_anchor; v_step := r.step_key; v_started := true;
    END IF;
    IF v_started THEN
      v_cursor := public.mos_work_due_at(v_cursor, r.target_hours);
    END IF;
  END LOOP;
  projected_ready_at := COALESCE(v_cursor, required_ready_at);
  risk_step := v_step;
  risk := CASE
    WHEN projected_ready_at <= required_ready_at THEN 'on_track'
    WHEN publish_at IS NULL OR projected_ready_at <= publish_at THEN 'at_risk'
    ELSE 'late' END;
  RETURN NEXT;
END $$;
```
Called by the planning sweep on each handoff to fill workflow_role_tasks.at_risk/risk_reason; replaces the coarse mos_plan_repair at_risk that reads the collapsed stage_deadlines date.
